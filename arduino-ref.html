Entradas y salidas

▼
Configurar pines
pinMode(pin, INPUT);          // pin como entrada
pinMode(pin, OUTPUT);         // pin como salida
pinMode(pin, INPUT_PULLUP);   // entrada con pull-up interno
Digital
digitalWrite(pin, HIGH);      // pone el pin en HIGH (5V)
digitalWrite(pin, LOW);       // pone el pin en LOW (0V)

int val = digitalRead(pin);   // lee HIGH o LOW
Analógico
int val = analogRead(pin);    // 0-1023 (resolución 10 bits)

analogWrite(pin, val);        // PWM: 0-255 (pines ~)

analogReference(DEFAULT);    // referencia voltaje ADC
// DEFAULT | INTERNAL | EXTERNAL
Tono
tone(pin, frecuencia);        // genera tono en Hz
tone(pin, frecuencia, durMs); // tono con duración
noTone(pin);                  // detiene tono
Pulsos
unsigned long dur = pulseIn(pin, HIGH);
// mide duración de un pulso en microsegundos
pulseIn(pin, HIGH, timeout);  // con timeout en µs

Control de tiempo

▼
Funciones de tiempo
delay(1000);                  // pausa 1000 ms (bloqueante)
delayMicroseconds(100);       // pausa 100 µs

unsigned long t = millis();  // ms desde inicio (desborda ~49 días)
unsigned long t = micros();  // µs desde inicio
Temporizador sin bloqueo (patrón)
unsigned long anterior = 0;
const long intervalo = 1000;  // ms

void loop() {
  unsigned long ahora = millis();
  if (ahora - anterior >= intervalo) {
    anterior = ahora;
    // código a ejecutar cada intervalo
  }
}

Comunicación Serial

▼
Configurar e imprimir

Serial.print(val, HEX);      // DEC | HEX | OCT | BIN
Serial.println(val, 4);      // 4 decimales (float)
Recibir datos
if (Serial.available() > 0) {  // bytes disponibles
  char c = Serial.read();      // lee 1 byte
  String s = Serial.readString();
  String s = Serial.readStringUntil('\n');
  int n = Serial.parseInt();    // lee entero
  float f = Serial.parseFloat();
}
Control y estado
Serial.flush();               // espera fin de transmisión

Serial.setTimeout(500);      // timeout en ms para reads

// Múltiples puertos (Mega/Leonardo)

Aritméticos y asignación

Comparación y lógicos
== != < > <= >=                         // comparación

Bitwise (bit a bit)
&    // AND bit a bit:   0b1100 & 0b1010 = 0b1000
|    // OR bit a bit:    0b1100 | 0b1010 = 0b1110
^    // XOR bit a bit:   0b1100 ^ 0b1010 = 0b0110
~    // NOT bit a bit:   ~0b1100 = 0b0011
<<   // shift izquierda: 1 << 3 = 8
>>   // shift derecha:   16 >> 2 = 4

x &= mask;  x |= mask;  x ^= mask;

Funciones de usuario

▼
Declarar y llamar

  return a + b;
}

void parpadear(int pin, int veces) {
  for (int i = 0; i < veces; i++) {
    digitalWrite(pin, HIGH);
    delay(200);
    digitalWrite(pin, LOW);
    delay(200);
  }
}

// Llamadas

parpadear(13, 5);

Interrupciones

▼
Configurar interrupción externa
volatile bool triggered = false;

void ISR_handler() {
  triggered = true;        // corta, sin Serial ni delay
}

attachInterrupt(
  digitalPinToInterrupt(2),  // pin (2 o 3 en Uno)
  ISR_handler,               // función ISR
  RISING                    // RISING|FALLING|CHANGE|LOW|HIGH
);

detachInterrupt(digitalPinToInterrupt(2));
Reglas de las ISR
// ✅ Variables compartidas deben ser volatile
// ✅ ISR debe ser corta y rápida
// ❌ No usar delay() dentro de ISR
// ❌ No usar Serial dentro de ISR
// ❌ millis() no avanza dentro de ISR

EEPROM

▼
Lectura y escritura básica
#include <EEPROM.h>

byte val = EEPROM.read(0);    // leer dirección 0
EEPROM.write(0, 42);          // escribir byte (lento, ~3.3ms)
EEPROM.update(0, 42);         // escribe solo si cambia (mejor)

int tamEEP = EEPROM.length(); // 1024 en Uno, 4096 en Mega
Guardar y leer tipos complejos
EEPROM.put(0, myFloat);       // guarda float (4 bytes)
EEPROM.put(4, myStruct);      // guarda struct completo

float f;
EEPROM.get(0, f);             // recupera float

Wire (I²C)

Wire.beginTransmission(0x3C);  // dirección esclavo
Wire.write(dato);

Servo

LiquidCrystal (LCD)

lcd.print("Hola!");
lcd.setCursor(0, 1);          // col 0, fila 1
lcd.clear();
